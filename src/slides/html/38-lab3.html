Un orquestador SAGA escrito como código

1 · Falla en ninguno → COMPLETADO

2 · 2 fallas temporales → reintentos a 1 s y 2 s

3 · Falla en CDN → compensa 2 pasos

4 · Falla en IA → agota reintentos, compensa 1

function* sagaProcesarLote(ctx) {

  for (const p of PASOS) {

    try { yield ctx.df.callActivityWithRetry(…);

      hechos.push(p.paso); }

    catch {

      for (const h of hechos.reverse())

        yield ctx.df.callActivity(compensar[h]);

      return 'COMPENSADO'; } }

  return 'COMPLETADO'; }
